## 1. SyncSessionController (lib/features/sync_zone/controllers/sync_session_controller.dart)

```dart
// Listens to Firestore syncSession doc in real time.
// Both conductor and listener read from this; only conductor writes.

import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';
import 'package:flutter/foundation.dart';
import '../../../core/api_client.dart';

enum SyncState { idle, playing, paused, ended }
enum SyncType  { radio, youtube, spotify }

class SyncSession {
  final SyncType  type;
  final String    contentId;
  final SyncState state;
  final int       positionMs;
  final String    conductorId;

  SyncSession({
    required this.type,
    required this.contentId,
    required this.state,
    required this.positionMs,
    required this.conductorId,
  });

  factory SyncSession.fromMap(Map<String, dynamic> data) => SyncSession(
    type:        SyncType.values.firstWhere((e) => e.name == (data['type'] ?? 'radio')),
    contentId:   data['contentId'] ?? '',
    state:       SyncState.values.firstWhere((e) => e.name == (data['state'] ?? 'idle')),
    positionMs:  data['positionMs'] ?? 0,
    conductorId: data['conductorId'] ?? '',
  );
}

class SyncSessionController extends ChangeNotifier {
  SyncSession? currentSession;
  bool get hasActiveSession =>
      currentSession != null && currentSession!.state != SyncState.idle;
  bool get isConductor =>
      currentSession?.conductorId == FirebaseAuth.instance.currentUser?.uid;

  StreamSubscription? _sub;

  void startListening(String coupleId) {
    _sub = FirebaseFirestore.instance
        .collection('couples').doc(coupleId)
        .collection('syncSession').doc('current')
        .snapshots()
        .listen((snap) {
      if (snap.exists && snap.data() != null) {
        currentSession = SyncSession.fromMap(snap.data()!);
        notifyListeners();
      }
    });
  }

  // Conductor: push new state to Firestore + backend (which notifies partner)
  Future<void> updateSession({
    required String   coupleId,
    required SyncType type,
    required String   contentId,
    required SyncState state,
    int positionMs = 0,
  }) async {
    await backendPost('/api/sync/session', {
      'type':       type.name,
      'contentId':  contentId,
      'state':      state.name,
      'positionMs': positionMs,
    });
  }

  Future<void> endSession() async {
    await backendPost('/api/sync/session', {'state': 'idle', 'type': 'radio', 'contentId': ''});
  }

  @override
  void dispose() {
    _sub?.cancel();
    super.dispose();
  }
}
```

## 3. Spotify OAuth Flow — NO manual key entry, ever

```dart
// SpotifyController (lib/features/sync_zone/controllers/spotify_controller.dart)
// 
// The flow:
// 1. Flutter calls backend → gets a Spotify login URL
// 2. Opens that URL in an InAppWebView (flutter_inappwebview)
// 3. User logs into Spotify in the webview (their own account, their own password)
// 4. Spotify redirects to your Replit backend callback
// 5. Backend saves tokens, redirects to heartsync://callback?spotify=connected
// 6. app_links package catches the deep link → closes webview → updates UI
//
// User NEVER sees an API key. User NEVER types credentials into HeartSync.
// They log into Spotify's own official login page, same as logging into Spotify app.

class SpotifyController extends ChangeNotifier {
  bool isConnected = false;
  bool isLoading   = false;
  Map<String, dynamic>? currentTrack;

  // Check if Spotify is already connected (token exists in Firestore)
  Future<void> checkConnection() async {
    try {
      final result = await backendGet('/api/spotify/current-track');
      isConnected = result['error'] == null;
    } catch (_) {
      isConnected = false;
    }
    notifyListeners();
  }

  // Step 1: get login URL from backend, open in webview
  Future<void> connectSpotify(BuildContext context) async {
    isLoading = true;
    notifyListeners();

    try {
      final result = await backendGet('/api/spotify/auth-url');
      final url    = result['url'] as String;

      // Open Spotify's official login page in an in-app webview
      // The webview closes automatically when app_links catches the deep link
      await Navigator.push(context, MaterialPageRoute(
        builder: (_) => SpotifyLoginWebView(loginUrl: url),
      ));

      // After webview closes, check if connection succeeded
      await checkConnection();
    } catch (e) {
      // show error snackbar
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Could not connect Spotify: $e'))
      );
    }
    isLoading = false;
    notifyListeners();
  }

  // Polls current track (conductor only, every 5 seconds)
  Future<void> fetchCurrentTrack() async {
    final result = await backendGet('/api/spotify/current-track');
    if (result['trackId'] != null) {
      currentTrack = result;
      notifyListeners();
    }
  }

  Future<void> disconnect() async {
    await backendPost('/api/spotify/disconnect', {});
    isConnected  = false;
    currentTrack = null;
    notifyListeners();
  }
}

// SpotifyLoginWebView widget:
// A full-screen InAppWebView loading the Spotify login URL.
// Intercepts URL changes — when it sees "heartsync://callback", 
// it closes itself (Navigator.pop) and app_links handles the rest.

class SpotifyLoginWebView extends StatefulWidget {
  final String loginUrl;
  // Build a full-screen Scaffold with:
  // - AppBar: "Connect Spotify" with a close X button
  // - InAppWebView loading loginUrl
  // - Override shouldOverrideUrlLoading: if url starts with "heartsync://",
  //   call Navigator.pop(context) and return NavigationActionPolicy.CANCEL
  //   so the webview doesn't try to navigate to the deep link itself
}
```

## 7. Deep link setup (app_links integration)

```dart
// In main.dart, initialise app_links to catch Spotify OAuth callbacks:

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  // ... existing Firebase init ...
  runApp(const HeartSyncApp());
}

// In your root widget's initState:
final _appLinks = AppLinks();

@override
void initState() {
  super.initState();
  _appLinks.uriLinkStream.listen((uri) {
    if (uri.scheme == 'heartsync' && uri.host == 'callback') {
      final spotify = uri.queryParameters['spotify'];
      if (spotify == 'connected') {
        // Close the SpotifyLoginWebView if it's still open
        // Update SpotifyController.isConnected = true
        // Show a success SnackBar: "Spotify connected! 🎵"
        context.read<SpotifyController>().checkConnection();
      }
    }
  });
}

// Android: add to AndroidManifest.xml inside <activity>:
// <intent-filter android:autoVerify="true">
//   <action android:name="android.intent.action.VIEW" />
//   <category android:name="android.intent.category.DEFAULT" />
//   <category android:name="android.intent.category.BROWSABLE" />
//   <data android:scheme="heartsync" android:host="callback" />
// </intent-filter>

// iOS: add to Info.plist:

// <array>
//   <dict>
```
